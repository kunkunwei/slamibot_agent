__COMPOSE__
services:
  core:
    image: registry.cn-shanghai.aliyuncs.com/slamibot/slamibot_d360_firmware:latest
    container_name: core
    network_mode: host
    privileged: true
    stdin_open: true
    tty: true
    restart: always
    environment:
      - TZ=Asia/Shanghai
    volumes:
      - /dev:/dev
      - type: bind
        source: /mnt

        bind:
          propagation: rshared
      - /etc/slamibot:/etc/slamibot
      - /etc/localtime:/etc/localtime:ro
    working_dir: /root/SLAMIBOT_D360_Framework
    command: bash -c "source install/setup.bash && roslaunch project_control core.launch"

  firmware-sensors:
    image: registry.cn-shanghai.aliyuncs.com/slamibot/slamibot_d360_firmware:latest

      - TZ=Asia/Shanghai
    volumes:
      - /dev:/dev
      - type: bind
        source: /mnt
        target: /mnt
        bind:
          propagation: rshared
      - /etc/slamibot:/etc/slamibot
      - /etc/localtime:/etc/localtime:ro
    working_dir: /root/SLAMIBOT_D360_Framework
