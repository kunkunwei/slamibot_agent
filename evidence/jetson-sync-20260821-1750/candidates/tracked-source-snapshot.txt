# Scout Mini 导航 Docker 入口
# 前置条件: 宿主机已启动 roscore + livox 雷达驱动 + /clock + CAN 底盘

set -e

# -------- ROS 环境 --------
source /opt/ros/noetic/setup.bash
source /Scout_mini_navigation/install/setup.bash
export ROS_PACKAGE_PATH=/Scout_mini_navigation/src:$ROS_PACKAGE_PATH


# 指向宿主机 roscore (host 网络模式下即本机)

echo "=========================================="
echo " ROS Master : $ROS_MASTER_URI"
echo " 工作空间   : /app"
echo "=========================================="

# -------- 1. rosbridge (WebSocket 19090, 非阻塞) --------
echo "[INFO] 启动 rosbridge_server (端口 19090)..."
rosrun rosbridge_server rosbridge_websocket __name:=scout_nav_rosbridge _port:=19090 _use_compression:=true > /tmp/rosbridge.log 2>&1 &
ROSBRIDGE_PID=$!
ROSBRIDGE_READY=0
for _ in $(seq 1 10); do
    if ! kill -0 "$ROSBRIDGE_PID" 2>/dev/null; then
        break
    fi
    if rosnode info /scout_nav_rosbridge >/dev/null 2>&1 && \
       timeout 1 bash -c 'exec 3<>/dev/tcp/127.0.0.1/19090' >/dev/null 2>&1; then
        ROSBRIDGE_READY=1
        break
    fi
    sleep 1
done
if [ "$ROSBRIDGE_READY" -eq 1 ]; then
    echo "[OK]   rosbridge 已启动 (PID=$ROSBRIDGE_PID)"
else
    echo "[ERROR] rosbridge 启动失败, 日志: /tmp/rosbridge.log"
    tail -n 50 /tmp/rosbridge.log || true
    exit 1
fi

# -------- 2. nav_multi_node (后台) --------
echo "[INFO] 启动 nav_multi_node ..."
rosrun nav_api nav_multi_node.py > /tmp/nav_multi.log 2>&1 &
NAV_PID=$!
sleep 1
if kill -0 "$NAV_PID" 2>/dev/null; then
    echo "[OK]   nav_multi_node 已启动 (PID=$NAV_PID)"
else
    echo "[ERROR] nav_multi_node 启动失败, 日志: /tmp/nav_multi.log"

    echo "[OK]   Web 网关已启动 (PID=$NGINX_PID)"

export function ModePanel() {
  const status = useRobotStore((s) => s.status);
  // 地图名仍由 robot store 提供(实时跟随机器人状态);模式判断交给 getLaunchStatus
  const mapName = status?.mapName ?? '';
  const [mapModal, setMapModal] = useState(false);

import { useCallback, useEffect, useState, Fragment } from 'react';
import { Stage, Layer, Image as KonvaImage, Circle, Line, RegularPolygon } from 'react-konva';
import type { KonvaEventObject } from 'konva/lib/Node';
import type { MapFrame } from '../../utils/mapProjection';
import { worldToCanvas, canvasToWorld } from '../../utils/mapProjection';
import type { NavPoint, NavArea, AreaVertex } from '../../types/map';
import { quaternionToYaw } from '../../utils/yaw';

export type CanvasMode = 'view' | 'addPoint' | 'drawArea';

export interface Map2DCanvasProps {
  frame: MapFrame | null;
  imageUrl: string | null;
  points: NavPoint[];
  areas: NavArea[];
  selectedPointId?: number | null;
  selectedAreaId?: number | null;
  robotPose?: { x: number; y: number; yaw: number } | null;
  mode: CanvasMode;
  onSelectPoint: (id: number) => void;
  onSelectArea: (id: number) => void;
  onCanvasClick: (world: { x: number; y: number; z: number }) => void;
  draftAreaVertices?: AreaVertex[];
  onDraftAreaVertex?: (v: AreaVertex) => void;
}

function useHtmlImage(url: string | null): HTMLImageElement | undefined {
  const [image, setImage] = useState<HTMLImageElement | undefined>(undefined);

  useEffect(() => {
    if (!url) {
      setImage(undefined);
      return;
    }

    const img = new window.Image();
    img.onload = () => setImage(img);
    img.src = url;

    return () => {
      img.onload = null;
    };
  }, [url]);

  return image;
}

function parseAreaVertices(pointsJson: string | null | undefined): AreaVertex[] {
  if (pointsJson == null || pointsJson === '') return [];
  try {
    const parsed = typeof pointsJson === 'string' ? JSON.parse(pointsJson) : pointsJson;
    if (!Array.isArray(parsed)) return [];
    return parsed.filter(
      (v): v is AreaVertex =>
        v != null
        && typeof v === 'object'
        && Number.isFinite((v as AreaVertex).x)
        && Number.isFinite((v as AreaVertex).y),
    );
  } catch {
    return [];
  }
}

function yawToKonvaRotation(yaw: number) {
  return (yaw * 180) / Math.PI - 90;
}

function getCanvasCoords(e: KonvaEventObject<MouseEvent | TouchEvent>) {
  const evt = e.evt as MouseEvent;
  if (typeof evt.offsetX === 'number' && typeof evt.offsetY === 'number') {
    return { x: evt.offsetX, y: evt.offsetY };
  }

  const pos = e.target.getStage()?.getPointerPosition();
  return pos ?? null;
}

export function Map2DCanvas(props: Map2DCanvasProps) {
  const {
    frame,
    imageUrl,
    points,
    areas,
    selectedPointId,
    selectedAreaId,
    robotPose,
    mode,
    onSelectPoint,
    onSelectArea,
    onCanvasClick,
    draftAreaVertices,
    onDraftAreaVertex,
  } = props;

  const mapImage = useHtmlImage(imageUrl);
  const stageWidth = frame?.width ?? 0;
  const stageHeight = frame?.height ?? 0;

  const handleStageClick = useCallback(
    (e: KonvaEventObject<MouseEvent | TouchEvent>) => {
      if (!frame || (mode !== 'addPoint' && mode !== 'drawArea')) return;

      const coords = getCanvasCoords(e);
      if (!coords) return;

      const world = canvasToWorld(coords.x, coords.y, frame);
      const vertex = { x: world.x, y: world.y, z: 0 };

      if (mode === 'drawArea') {
        onDraftAreaVertex?.(vertex);
      }

      onCanvasClick(vertex);
    },
    [frame, mode, onCanvasClick, onDraftAreaVertex],
  );

  if (!frame) {
    return (
      <div
        data-testid="map-2d-canvas"
        style={{
          height: '100%',
          minHeight: 320,
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'center',
          color: 'rgba(0, 0, 0, 0.45)',
          background: '#fafafa',
          fontSize: 14,
          padding: 24,
          textAlign: 'center',
        }}
      >
        等待 2D 底图（需 rosbridge 订阅 /map）。未连接机器狗时列表仍可浏览，但无法在画布上标注。
      </div>
    );
  }

  return (
    <div style={{ overflow: 'auto' }}>
      <div data-testid="map-2d-canvas">
        <Stage
          width={stageWidth}
          height={stageHeight}
          onClick={handleStageClick}
          onTap={handleStageClick}
        >
          <Layer>
            {mapImage && frame && (
              <KonvaImage image={mapImage} width={frame.width} height={frame.height} />
            )}

            {frame &&
              areas.map((area) => {
                const flat = parseAreaVertices(area.points).flatMap((v) => {
                  const { x, y } = worldToCanvas(v.x, v.y, frame);
                  return [x, y];
                });

                return (
                  <Line
                    key={area.id}
                    points={flat}
                    closed
                    fill="rgba(255,0,0,0.2)"
                    stroke={selectedAreaId === area.id ? '#ff0000' : '#cc0000'}
                    strokeWidth={selectedAreaId === area.id ? 2 : 1}
                    onClick={() => onSelectArea(area.id)}
                    onTap={() => onSelectArea(area.id)}
                  />
                );
              })}

            {frame && draftAreaVertices && draftAreaVertices.length > 0 && (
              <Line
                points={draftAreaVertices.flatMap((v) => {
                  const { x, y } = worldToCanvas(v.x, v.y, frame);
                  return [x, y];
                })}
                stroke="#ff6600"
                strokeWidth={2}
                dash={[6, 4]}
              />
            )}

            {frame &&
              points.map((point) => {
                const { x, y } = worldToCanvas(point.positionX, point.positionY, frame);
                const yaw = quaternionToYaw({
                  x: point.orientationX,
                  y: point.orientationY,
                  z: point.orientationZ,
                  w: point.orientationW,
                });
                const selected = selectedPointId === point.id;

                return (
                  <Fragment key={point.id}>
                    <Circle
                      x={x}
                      y={y}
                      radius={selected ? 8 : 6}
                      fill={selected ? '#ff4d4f' : '#52c41a'}
                      stroke="#fff"
                      strokeWidth={1}
                      onClick={() => onSelectPoint(point.id)}
                      onTap={() => onSelectPoint(point.id)}
                    />
                    <RegularPolygon
                      x={x}
                      y={y}
                      sides={3}
                      radius={10}
                      fill={selected ? '#ff4d4f' : '#52c41a'}
                      rotation={yawToKonvaRotation(yaw)}
                      onClick={() => onSelectPoint(point.id)}
                      onTap={() => onSelectPoint(point.id)}
                    />
                  </Fragment>
                );
              })}

            {frame && robotPose && (() => {
              const { x, y } = worldToCanvas(robotPose.x, robotPose.y, frame);
              return (
                <RegularPolygon
                  x={x}
                  y={y}
                  sides={3}
                  radius={12}
                  fill="#1890ff"
                  rotation={yawToKonvaRotation(robotPose.yaw)}
                />
              );
            })()}
          </Layer>
        </Stage>
      </div>
    </div>
  );
}

export function PointsPage() {
  const statusQuery = useRobotStatusQuery();
  useRobotPose();
  const { data: maps = [] } = useQuery({
    queryKey: ['maps'],
    queryFn: getMapList,
    select: normalizeMaps,
  });

  const statusMapName = statusQuery.data?.mapName ?? '';
  const defaultMapName = statusMapName || maps[0]?.mapName || '';
  const [mapNameOverride, setMapNameOverride] = useState<string | null>(null);
  const mapName = mapNameOverride ?? defaultMapName;

  const { frame, imageUrl } = useOccupancyGrid();
  const robotPose = useRobotPoseYaw();
  const pointsQuery = usePointsQuery(mapName);
  const areasQuery = useAreasQuery(mapName);

  const [canvasMode, setCanvasMode] = useState<CanvasMode>('view');
  const [draftAreaVertices, setDraftAreaVertices] = useState<AreaVertex[]>([]);
  const [selectedPointId, setSelectedPointId] = useState<number | null>(null);
  const [selectedAreaId, setSelectedAreaId] = useState<number | null>(null);

  const [pointModalOpen, setPointModalOpen] = useState(false);
  const [initialPosition, setInitialPosition] = useState<{ x: number; y: number; z: number } | undefined>();

  const [areaModalOpen, setAreaModalOpen] = useState(false);

  const resetDrawState = useCallback(() => {
    setCanvasMode('view');
    setDraftAreaVertices([]);
  }, []);

  const ensureCanAnnotate = () => {
    if (!mapName) {
      message.warning('请先选择地图');
      return false;
    }
    if (!frame) {
      message.warning('暂无 2D 底图（需 ROS /map），无法在画布标注');
      return false;
    }
    return true;
  };

  const handleAddPointMode = () => {
    if (!ensureCanAnnotate()) return;
    setCanvasMode('addPoint');
    setDraftAreaVertices([]);
  };

  const handleDrawAreaMode = () => {
    if (!ensureCanAnnotate()) return;
    setCanvasMode('drawArea');
    setDraftAreaVertices([]);
  };

  const handleCanvasClick = (world: { x: number; y: number; z: number }) => {
    if (canvasMode === 'addPoint') {
      setInitialPosition(world);
      setPointModalOpen(true);
      setCanvasMode('view');
    }
  };

  const handleDraftAreaVertex = (v: AreaVertex) => {
    setDraftAreaVertices((prev) => [...prev, v]);
  };

  const finishAreaDraw = () => {
    if (draftAreaVertices.length >= 3) {
      setAreaModalOpen(true);
      setCanvasMode('view');
    }
  };

  const handlePointModalClose = () => {
    setPointModalOpen(false);
    setInitialPosition(undefined);
  };

  const handleAreaModalClose = () => {
    setAreaModalOpen(false);
    setDraftAreaVertices([]);
  };

  return (
    <div style={{ display: 'flex', height: '100%', background: '#fff', color: 'rgba(0, 0, 0, 0.88)' }}>
      <div style={{ width: 360, borderRight: '1px solid #f0f0f0', overflow: 'auto', background: '#fff' }}>
        <PointsSidebar
          mapName={mapName}
          onMapChange={setMapNameOverride}
          selectedPointId={selectedPointId}
          onSelectPoint={setSelectedPointId}
          selectedAreaId={selectedAreaId}
          onSelectArea={setSelectedAreaId}
          onAddPointMode={handleAddPointMode}
          onDrawAreaMode={handleDrawAreaMode}
          onRefresh={() => {
            void pointsQuery.invalidate();
            void areasQuery.invalidate();
          }}
        />
      </div>
      <div style={{ flex: 1, position: 'relative', background: '#fff' }}>
        <Map2DCanvas
          frame={frame}
          imageUrl={imageUrl}
          points={pointsQuery.data ?? []}
          areas={areasQuery.data ?? []}
          selectedPointId={selectedPointId}
          selectedAreaId={selectedAreaId}
          robotPose={robotPose}
          mode={canvasMode}
          onSelectPoint={setSelectedPointId}
          onSelectArea={setSelectedAreaId}
          onCanvasClick={handleCanvasClick}
          draftAreaVertices={draftAreaVertices}
          onDraftAreaVertex={handleDraftAreaVertex}
        />
        {canvasMode !== 'view' && (
          <div style={{ position: 'absolute', top: 8, left: 8 }}>
            <Space>
              <Button onClick={resetDrawState}>取消</Button>
              {canvasMode === 'drawArea' && (

                </Button>
              )}

          onSuccess={() => void areasQuery.invalidate()}

  );
}

-- nav_api 数据库表结构(MVP:地图表 + 点位表)
-- 说明:SQLite 为动态类型,String(N)/Float(15) 等长度精度约束在应用层校验,
--      此处仅保留数据库层能强制的约束(主键/唯一/外键/CHECK/默认值)。

-- ============================================================
-- Map 地图表
-- 点位/任务的 mapName 外键指向本表;本项目为纯 2D,mapType 用 2d_grid。
-- ============================================================
CREATE TABLE IF NOT EXISTS Map (
    id             INTEGER PRIMARY KEY AUTOINCREMENT,
    mapName        TEXT    NOT NULL UNIQUE,                       -- 唯一,且不能为 'default'
    mapDisplayName TEXT,                                          -- 显示名(支持中文)
    mapDescription TEXT,                                          -- 地图描述
    mapType        TEXT    NOT NULL DEFAULT '2d_grid',            -- 3d_pointcloud / 2d_grid
    pcdFilePath    TEXT,                                          -- 3D 点云路径(本项目恒空)
    yamlFilePath   TEXT,                                          -- 2D 地图 .yaml 路径(关键)
    isActive       INTEGER NOT NULL DEFAULT 0,                    -- 1=当前激活地图(导航栈使用),0=未激活;通过 map/switch 切换
    createdTime    TEXT    NOT NULL DEFAULT (datetime('now','localtime')),
    updatedTime    TEXT    NOT NULL DEFAULT (datetime('now','localtime')),
    sort           INTEGER NOT NULL DEFAULT 0,
    CHECK (mapName <> 'default')
);

-- ============================================================
-- PointPosition 点位表
-- 同一地图下 pointName 唯一;坐标为 map 系(米),2D 场景 z=0、姿态仅 z/w。
-- ============================================================
CREATE TABLE IF NOT EXISTS PointPosition (
    id                  INTEGER PRIMARY KEY AUTOINCREMENT,
    type                INTEGER NOT NULL,                         -- PointType 枚举(巡检点=2)
    mapName             TEXT    NOT NULL,                         -- 关联 Map.mapName
    pointName           TEXT    NOT NULL,                         -- 同图内唯一
    positionX           REAL    NOT NULL,
    positionY           REAL    NOT NULL,
    positionZ           REAL    NOT NULL DEFAULT 0,
    orientationX        REAL    NOT NULL DEFAULT 0,
    orientationY        REAL    NOT NULL DEFAULT 0,
    orientationZ        REAL    NOT NULL DEFAULT 0,
    orientationW        REAL    NOT NULL DEFAULT 1,
    controlIp           TEXT,                                     -- 以下门控字段 MVP 留空
    controlPort         INTEGER,
    controlType         INTEGER,
    controlAddress      INTEGER,
    controlDelayClosure INTEGER,
    action              TEXT    DEFAULT '',
    actionContent       TEXT    DEFAULT '',
    sort                INTEGER NOT NULL DEFAULT 0,
    UNIQUE (mapName, pointName),
    FOREIGN KEY (mapName) REFERENCES Map(mapName)
);

-- ============================================================
-- TaskFlow 任务表(一条固定巡检路线)
-- 一个任务含 N 个有序点位(TaskPoint);删任务级联删其点位行,不碰 PointPosition。
-- ============================================================
CREATE TABLE IF NOT EXISTS TaskFlow (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    taskName    TEXT    NOT NULL,                            -- 同一地图下唯一
    description TEXT,
    mapName     TEXT    NOT NULL,                            -- 关联 Map.mapName
    isEnabled   INTEGER NOT NULL DEFAULT 1,                  -- 0/1 是否启用
    createdTime TEXT    NOT NULL DEFAULT (datetime('now','localtime')),
    updatedTime TEXT    NOT NULL DEFAULT (datetime('now','localtime')),
    UNIQUE (mapName, taskName),
    FOREIGN KEY (mapName) REFERENCES Map(mapName)
);

-- ============================================================
-- TaskPoint 任务点位表(引用模型)
-- 非临时点位:靠 pointId 软引用 PointPosition(不加强制外键),执行/展示时实时查;
--             点位被删则该站"失效",执行时报错。
-- 临时点位  :pointId 为空,坐标内联存本表。
-- ============================================================
CREATE TABLE IF NOT EXISTS TaskPoint (
    id            INTEGER PRIMARY KEY AUTOINCREMENT,
    taskId        INTEGER NOT NULL,                          -- 关联 TaskFlow.id
    pointId       INTEGER,                                   -- 软引用 PointPosition.id;临时点位为 NULL
    pointName     TEXT    NOT NULL,                          -- 冗余:展示 + 点位被删后报错用
    type          INTEGER NOT NULL DEFAULT 2,                -- PointType 枚举
    pointOrder    INTEGER NOT NULL DEFAULT 0,                -- 执行顺序(避开 SQL 保留字 order)
    isTemporary   INTEGER NOT NULL DEFAULT 0,                -- 0/1 是否临时点位
    positionX     REAL,                                      -- 临时点位必填;非临时点位为空(实时查)
    positionY     REAL,
    positionZ     REAL,
    orientationX  REAL,
    orientationY  REAL,
    orientationZ  REAL,
    orientationW  REAL,
    action        TEXT    DEFAULT '',
    actionContent TEXT    DEFAULT '',
    FOREIGN KEY (taskId) REFERENCES TaskFlow(id) ON DELETE CASCADE
);

-- ============================================================
-- Area 区域表(电子围栏 / 巡检多边形)
-- 多边形顶点序列以 JSON 字符串存储,与前端 API 契约一致;
-- 同图内 areaName 唯一。
-- ============================================================
CREATE TABLE IF NOT EXISTS Area (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    areaName    TEXT    NOT NULL,
    mapName     TEXT    NOT NULL,
    description TEXT,
    type        INTEGER NOT NULL DEFAULT 0,
    points      TEXT    DEFAULT '[]',
    createdTime TEXT    NOT NULL DEFAULT (datetime('now','localtime')),
    updatedTime TEXT    NOT NULL DEFAULT (datetime('now','localtime')),
    UNIQUE (mapName, areaName),
    FOREIGN KEY (mapName) REFERENCES Map(mapName)
);

-- ============================================================
-- CycleConfig 周期执行配置表
-- 关联到 TaskFlow.id,描述一个任务以何种周期/次数重复执行。
-- cycleMode:0=按次,1=按日(cron 风格 HH:MM)。
-- ============================================================
CREATE TABLE IF NOT EXISTS CycleConfig (
    id            INTEGER PRIMARY KEY AUTOINCREMENT,
    taskId        INTEGER NOT NULL,
    cycleMode     INTEGER NOT NULL DEFAULT 0,
    cycleInterval INTEGER NOT NULL DEFAULT 0,         -- 按次模式:每次间隔秒数
    cycleCount    INTEGER NOT NULL DEFAULT 0,         -- 0=无限;>0=本周期内最大执行次数
    startTime     TEXT,                                -- ISO 字符串,首次执行时间
    dailyTime     TEXT,                                -- 仅 cycleMode=1 时使用,形如 HH:MM
    isRunning     INTEGER NOT NULL DEFAULT 0,         -- 0/1 当前是否启用
    nextExecuteTime TEXT,                              -- 下次计划执行时间(ISO)
    createdTime   TEXT NOT NULL DEFAULT (datetime('now','localtime')),
    updatedTime   TEXT NOT NULL DEFAULT (datetime('now','localtime')),
    FOREIGN KEY (taskId) REFERENCES TaskFlow(id) ON DELETE CASCADE
);

def _normalize(x: float, y: float, z: float, w: float):
    norm = (x * x + y * y + z * z + w * w) ** 0.5
    if norm < 1e-6:
        raise ValueError("四元数模长接近 0,非法姿态")
    return x / norm, y / norm, z / norm, w / norm

        )

        request.positionX,

        request.positionZ,

@router.post("/save", response_model=ApiResponse)
def save_map(request: MapSaveRequest) -> ApiResponse:
    """停止 FAST_LIO 建图并保存 PCD 与二维栅格地图。"""
    if request.mapName == "default":
        return fail("mapName 非法或为保留名 default")
    if not process_manager.get_status().get("mapping_running"):
        return fail("当前未在建图模式,无法保存地图")

    conn = get_conn()
    try:
        if conn.execute(
            "SELECT 1 FROM Map WHERE mapName = ?", (request.mapName,)
        ).fetchone():
            return fail("地图已存在: %s" % request.mapName)
        stopped, message = process_manager.stop_mapping()
        if not stopped:
            return fail("停止建图进程失败: %s" % message)
        if not os.path.exists(_FASTLIO_PCD_FILE):
            return fail("FAST_LIO 未生成 PCD 文件,可能建图时间太短或点云为空")

        target_dir = os.path.join(_MAP_ROOT, request.mapName)
        os.makedirs(target_dir, exist_ok=True)
        target_pcd = os.path.join(target_dir, request.mapName + ".pcd")
        shutil.copy2(_FASTLIO_PCD_FILE, target_pcd)

        prefix = os.path.join(target_dir, request.mapName)
        yaml_path: Optional[str] = prefix + ".yaml"
        try:
            result = subprocess.run(
                [
                    "python3",
                    _PCD_TO_MAP,
                    target_pcd,
                    "-o",
                    prefix,
                    "--fill-free",
                ],
                capture_output=True,
                text=True,
                timeout=120,
                cwd=os.path.dirname(_PCD_TO_MAP),
            )
            if result.returncode != 0 or not os.path.exists(yaml_path):
                logging.warning("pcd_to_map failed: %s", result.stderr[:500])
                yaml_path = None
        except (subprocess.TimeoutExpired, FileNotFoundError) as exc:
            logging.warning("pcd_to_map skipped: %s", exc)

@router.post("/task/execute", response_model=ApiResponse)
def nav_execute(request: NavExecuteRequest) -> ApiResponse:
    """执行数据库中保存的多点导航任务。"""
    conflict = _check_conflict()
    if conflict:
        return fail(conflict)

    with get_conn() as conn:
        task = conn.execute(
            "SELECT id, taskName, mapName, isEnabled FROM TaskFlow WHERE id = ?",
            (request.taskId,),
        ).fetchone()
        if task is None:
            return fail("任务不存在: taskId=%s" % request.taskId)
        if not task["isEnabled"]:
            return fail("任务已禁用,无法执行")

        rows = conn.execute(
            "SELECT tp.id, tp.pointName, tp.isTemporary, tp.pointId, tp.pointOrder, "
            "tp.positionX, tp.positionY, tp.positionZ, "
            "tp.orientationX, tp.orientationY, tp.orientationZ, tp.orientationW, "
            "pp.positionX AS pp_x, pp.positionY AS pp_y, pp.positionZ AS pp_z, "
            "pp.orientationX AS pp_ox, pp.orientationY AS pp_oy, "
            "pp.orientationZ AS pp_oz, pp.orientationW AS pp_ow "
            "FROM TaskPoint tp LEFT JOIN PointPosition pp ON pp.id = tp.pointId "
            "WHERE tp.taskId = ? ORDER BY tp.pointOrder, tp.id",
            (request.taskId,),
        ).fetchall()
        if not rows:
            return fail("任务中没有点位")

        points = []
        invalid = []
        for row in rows:
            if row["isTemporary"]:
                prefix = ""
            elif row["pp_x"] is None:
                invalid.append(row["pointName"])
                continue
            else:
                prefix = "pp_"
            points.append(
                {
                    "pointName": row["pointName"],
                    "positionX": row[prefix + "x"] if prefix else row["positionX"],
                    "positionY": row[prefix + "y"] if prefix else row["positionY"],
                    "positionZ": (
                        row[prefix + "z"] if prefix else row["positionZ"]
                    )
                    or 0.0,
                    "orientationX": (
                        row[prefix + "ox"] if prefix else row["orientationX"]
                    )
                    or 0.0,
                    "orientationY": (
                        row[prefix + "oy"] if prefix else row["orientationY"]
                    )
                    or 0.0,
                    "orientationZ": (
                        row[prefix + "oz"] if prefix else row["orientationZ"]
                    )
                    or 0.0,
                    "orientationW": (
                        row[prefix + "ow"] if prefix else row["orientationW"]
                    )
                    if (
                        row[prefix + "ow"] if prefix else row["orientationW"]
                    )
                    is not None
                    else 1.0,
                }
            )
        if invalid:
            return fail("以下点位已不存在,请先修正任务: %s" % "、".join(invalid))

    return _send_task(task["id"], task["taskName"], points)

"""ROS 1 access from Python 3.10+ through rosbridge/roslibpy."""

import logging
import os
import json
import threading
from typing import Any, Optional, Tuple

import roslibpy


LOGGER = logging.getLogger(__name__)


class RosbridgeClient:
    """Keep a rosbridge connection and cache the latest AMCL pose."""

    def __init__(self) -> None:
        self._ros = roslibpy.Ros(
            host=os.environ.get("ROSBRIDGE_HOST", "127.0.0.1"),
            port=int(os.environ.get("ROSBRIDGE_PORT", "19090")),
        )
        self._pose_topic: Optional[roslibpy.Topic] = None
        self._topics: list[roslibpy.Topic] = []
        self._transport_running = False
        self._latest_pose: Optional[dict[str, float]] = None
        self._latest_nav_status: dict[str, Any] = {}
        self._latest_scout_status: Optional[dict[str, Any]] = None
        self._latest_bms_status: Optional[dict[str, Any]] = None
        self._latest_battery: Optional[dict[str, Any]] = None
        self._cmd_vel_topic: Optional[roslibpy.Topic] = None
        self._teleop_topic: Optional[roslibpy.Topic] = None
        self._lock = threading.Lock()

    @property
    def is_connected(self) -> bool:
        return bool(self._ros.is_connected)

    def start(self) -> None:
        """Connect without preventing HTTP startup when ROS is unavailable."""
        try:
            self._ros.run(timeout=3)
            self._transport_running = True
        except Exception as exc:
            LOGGER.warning("rosbridge unavailable; ROS endpoints start degraded: %s", exc)
            self._ros.terminate()
            return

        self._pose_topic = roslibpy.Topic(
            self._ros,
            "/amcl_pose",
            "geometry_msgs/PoseWithCovarianceStamped",
        )
        self._pose_topic.subscribe(self._on_pose)
        self._topics = [
            self._pose_topic,
            self._subscribe(
                "/nav_multi/status", "std_msgs/String", self._on_nav_status
            ),
            self._subscribe(
                "/scout_status", "scout_msgs/ScoutStatus", self._on_scout_status
            ),
            self._subscribe(
                "/BMS_status", "scout_msgs/ScoutBmsStatus", self._on_bms_status
            ),
            self._subscribe(
                "/battery", "sensor_msgs/BatteryState", self._on_battery
            ),
        ]
        LOGGER.info("connected to rosbridge")

    def stop(self) -> None:
        for topic in self._topics:
            topic.unsubscribe()
        with self._lock:
            if self._teleop_topic is not None:
                try:
                    self._teleop_topic.unsubscribe()
                except Exception as exc:
                    LOGGER.warning("ros_client: 取消 teleop 订阅失败: %s", exc)

def _normalize(x: float, y: float, z: float, w: float):
    norm = (x * x + y * y + z * z + w * w) ** 0.5
    if norm < 1e-6:
        raise ValueError("四元数模长接近 0,非法姿态")
    return x / norm, y / norm, z / norm, w / norm


def _prepare_point(
    index: int, point: TaskPointInput
) -> Union[tuple, str]:
    order = point.order if point.order is not None else index
    if point.isTemporary:
        coordinates = (
            point.positionX,
            point.positionY,
            point.positionZ,
            point.orientationX,
            point.orientationY,
            point.orientationZ,
            point.orientationW,
        )
        if any(value is None for value in coordinates):
            return "临时点位 points[%d] 缺少坐标或姿态" % index
        try:
            qx, qy, qz, qw = _normalize(
                point.orientationX,
                point.orientationY,
                point.orientationZ,
                point.orientationW,
            )
        except (ValueError, TypeError) as exc:
            return "临时点位 points[%d] 姿态非法: %s" % (index, exc)
        return (
            None,
            point.pointName,
            point.type,
            order,
            1,
            point.positionX,
            point.positionY,
            point.positionZ,
            qx,
            qy,
            qz,
            qw,
            point.action,
            point.actionContent,
        )
    if point.pointId is None:
        return "非临时点位 points[%d] 缺少 pointId" % index
    return (
        point.pointId,
        point.pointName,
        point.type,
        order,
        0,
        None,
        None,
        None,
        None,
        None,
        None,
        None,
        point.action,
        point.actionContent,
    )


@router.post("/save", response_model=ApiResponse)
def task_save(request: TaskSaveRequest) -> ApiResponse:
    """保存一条由有序点位组成的导航任务。"""
    conn = get_conn()
    try:
        if conn.execute(
            "SELECT 1 FROM Map WHERE mapName = ?", (request.mapName,)
        ).fetchone() is None:
            return fail("地图不存在: %s" % request.mapName)
        if conn.execute(
            "SELECT 1 FROM TaskFlow WHERE mapName = ? AND taskName = ?",
            (request.mapName, request.taskName),
        ).fetchone() is not None:
            return fail("该地图下已存在同名任务: %s" % request.taskName)

        prepared = []
        for index, point in enumerate(request.points):
            row = _prepare_point(index, point)
            if isinstance(row, str):
                return fail(row)
            prepared.append(row)

        cur = conn.execute(
            "INSERT INTO TaskFlow (taskName, description, mapName, isEnabled) "
            "VALUES (?,?,?,?)",
            (
                request.taskName,
                request.description,
                request.mapName,
                1 if request.isEnabled else 0,
            ),
        )
        task_id = cur.lastrowid
        conn.executemany(
            "INSERT INTO TaskPoint (%s) VALUES (%s)"
            % (_TP_COLUMNS, ",".join(["?"] * 15)),
            [(task_id,) + row for row in prepared],
        )
        conn.commit()
        return ok({"taskId": task_id}, "保存任务成功")
    except Exception as exc:
        conn.rollback()
        return fail("保存任务失败: %s" % exc)
    finally:
        conn.close()


def _resolve_point(conn, task_point: dict) -> dict:
    if task_point["isTemporary"]:
        task_point["valid"] = True
        return task_point
    point = conn.execute(
        "SELECT positionX, positionY, positionZ, "
        "orientationX, orientationY, orientationZ, orientationW "
        "FROM PointPosition WHERE id = ?",
        (task_point["pointId"],),
    ).fetchone()
    if point is None:
        task_point["valid"] = False
    else:
        task_point["valid"] = True
        task_point.update(dict(point))
    return task_point


@router.get("/list", response_model=ApiResponse)
def task_list(request: Annotated[TaskListRequest, Query()]) -> ApiResponse:
    """查询任务列表，并解析每个任务的实时点位信息。"""
    conn = get_conn()
    try:
        if request.mapName:
            tasks = conn.execute(
                "SELECT * FROM TaskFlow WHERE mapName = ? ORDER BY id",
                (request.mapName,),
            ).fetchall()
        else:
            tasks = conn.execute(
                "SELECT * FROM TaskFlow ORDER BY id"
            ).fetchall()

        result = []
        for task_row in tasks:
            task = dict(task_row)
            task_points = conn.execute(
                "SELECT * FROM TaskPoint WHERE taskId = ? ORDER BY pointOrder, id",
                (task["id"],),
            ).fetchall()
            task["points"] = [
                _resolve_point(conn, dict(point)) for point in task_points
            ]
            result.append(task)
        return ok(result)
    finally:
        conn.close()


@router.post("/update", response_model=ApiResponse)
def task_update(request: TaskUpdateRequest) -> ApiResponse:
    """部分更新任务；传 points 时完全替换任务点位。"""
    data = request.model_dump(exclude_unset=True)
    task_id = data.pop("id")
    points = data.pop("points", None)

    conn = get_conn()
    try:
        row = conn.execute(
            "SELECT * FROM TaskFlow WHERE id = ?", (task_id,)
        ).fetchone()
        if row is None:
            return fail("任务不存在: id=%s" % task_id)
        current = dict(row)

        if "taskName" in data and data["taskName"] != current["taskName"]:
            conflict = conn.execute(
                "SELECT 1 FROM TaskFlow "
                "WHERE mapName = ? AND taskName = ? AND id != ?",
                (current["mapName"], data["taskName"], task_id),
            ).fetchone()
            if conflict:
                return fail("该地图下已存在同名任务: %s" % data["taskName"])

        prepared = None
        if points is not None:
            if not points:
                return fail("points 至少包含 1 个点位")
            prepared = []
            for index, raw_point in enumerate(points):
                result = _prepare_point(index, TaskPointInput.model_validate(raw_point))
                if isinstance(result, str):
                    return fail(result)
                prepared.append(result)

        if data:
            fields = ", ".join("%s = ?" % field for field in data)
            fields += ", updatedTime = datetime('now','localtime')"
            conn.execute(
                "UPDATE TaskFlow SET %s WHERE id = ?" % fields,
                list(data.values()) + [task_id],
            )
        if prepared is not None:
            conn.execute("DELETE FROM TaskPoint WHERE taskId = ?", (task_id,))
            conn.executemany(
                "INSERT INTO TaskPoint (%s) VALUES (%s)"
                % (_TP_COLUMNS, ",".join(["?"] * 15)),
                [(task_id,) + point for point in prepared],
            )
        conn.commit()
        return ok({"taskId": task_id}, "更新任务成功")
    except Exception as exc:
        conn.rollback()
        return fail("更新任务失败: %s" % exc)
    finally:
        conn.close()

@router.post("/cycle-config", response_model=ApiResponse)
def cycle_config(request: CycleConfigRequest) -> ApiResponse:
    """为指定任务创建/替换周期执行配置。

    同一个 taskId 只允许一条 isRunning=1 的配置；新提交时将旧的 isRunning=0。

    conn = get_conn()
    try:
        if conn.execute(
            "SELECT 1 FROM TaskFlow WHERE id = ?", (request.taskId,)
        ).fetchone() is None:
            return fail("任务不存在: taskId=%s" % request.taskId)

        conn.execute(
            "UPDATE CycleConfig SET isRunning = 0 WHERE taskId = ?",
            (request.taskId,),
        )
        next_time = _compute_next_execute(
            request.cycleMode,
            request.startTime or request.dailyTime,
            request.cycleInterval,
        )
        cur = conn.execute(
            """INSERT INTO CycleConfig
               (taskId, cycleMode, cycleInterval, cycleCount,
                startTime, dailyTime, isRunning, nextExecuteTime)
               VALUES (?, ?, ?, ?, ?, ?, 1, ?)""",
            (
                request.taskId,
                request.cycleMode,
                request.cycleInterval,
                request.cycleCount,
                request.startTime,
                request.dailyTime,
                next_time,
            ),
        )
        conn.commit()
        return ok({"configId": cur.lastrowid}, "周期配置已创建")
    except Exception as exc:
        conn.rollback()
        return fail("创建周期配置失败: %s" % exc)
    finally:
        conn.close()
